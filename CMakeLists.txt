cmake_minimum_required(VERSION 3.18)
project(thermal_twin LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(thermal_engine cpp/main.cpp cpp/thermal_cpu.cpp)
target_include_directories(thermal_engine PRIVATE cpp)
find_package(OpenMP)
if(OpenMP_CXX_FOUND)
  target_link_libraries(thermal_engine PRIVATE OpenMP::OpenMP_CXX)
  target_compile_definitions(thermal_engine PRIVATE HAVE_OPENMP=1)
endif()
include(CheckLanguage)
check_language(CUDA)
if(CMAKE_CUDA_COMPILER)
  enable_language(CUDA)
  target_sources(thermal_engine PRIVATE cpp/thermal_cuda.cu)
  set_target_properties(thermal_engine PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  target_compile_definitions(thermal_engine PRIVATE HAVE_CUDA=1)
endif()
enable_testing()
add_test(NAME engine_help COMMAND thermal_engine --help)
